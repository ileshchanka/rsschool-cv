Code Example

My solution to the Valid Parentheses kata (5 kyu) on Codewars. A single pass over the string with a counter is enough: the balance may never go negative, and it has to end at zero.

function validParentheses(string) {
  let balance = 0;

  for (const char of string) {
    if (char === '(') {
      balance += 1;
    } else if (char === ')') {
      balance -= 1;
    }

    if (balance < 0) {
      return false;
    }
  }

  return balance === 0;
}
